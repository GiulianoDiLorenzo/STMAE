def load_audio(audio_path, start_time, end_time, sr=44100):
    """
    Load a mono audio segment from file between start_time and end_time.
    Returns a 1D NumPy array of samples.
    """
    duration = end_time - start_time
    y, _ = librosa.load(audio_path, sr=sr, mono=True, offset=start_time, duration=duration)
    return y

# 🔧 You must implement this
def compute_constant_q(audio, sr=44100):
    """
    Compute the Constant-Q Transform spectrogram.
    Returns a (T, F) array. Recommended shape: (128, 96)
    """

    log_cqt = log_cqt.T  # shape becomes (time, frequency)
